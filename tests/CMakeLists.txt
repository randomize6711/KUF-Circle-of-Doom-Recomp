# --- Instruction semantics -------------------------------------------------------------------------
# kuf_add_insn_suite(<target> <asm_dir> <test prefix>): assemble <asm_dir>/*.s in the sandbox, recompile them
# with rexglue recompile-tests and build a Catch2 executable from the result.
function(kuf_add_insn_suite target asm_dir prefix)
    cmake_parse_arguments(ARG "EXPECT_FAIL" "" "" ${ARGN})
    set(work ${CMAKE_BINARY_DIR}/insn-tests/${target})
    file(GLOB asm_files ${asm_dir}/*.s)
    add_custom_command(
        OUTPUT ${work}/generated/ppc_test_functions.cpp ${work}/generated/ppc_test_cases.cpp
               ${work}/generated/ppc_test_decls.h
        COMMAND ${CMAKE_COMMAND} -E rm -rf ${work}/bin ${work}/obj
        COMMAND ${CMAKE_SOURCE_DIR}/tools/ppc_asm_sandboxed.sh ${work} ${asm_dir}
        COMMAND $<TARGET_FILE:rex::rexglue> recompile-tests --output=${work}/generated
                --bin-dir=${work}/bin --asm-dir=${asm_dir}
        DEPENDS ${asm_files} ${CMAKE_SOURCE_DIR}/tools/ppc_asm_sandboxed.sh rex::rexglue
        COMMENT "Assembling (sandboxed) and recompiling ${target}"
        VERBATIM)
    add_executable(${target} ${work}/generated/ppc_test_functions.cpp ${work}/generated/ppc_test_cases.cpp)
    target_include_directories(${target} PRIVATE ${work}/generated ${REXSDK_DIR}/include)
    target_link_libraries(${target} PRIVATE simde rexruntime fmt::fmt Catch2::Catch2WithMain)
    target_compile_options(${target} PRIVATE ${KUFCOD_TEST_COPTS} -include limits)
    set_target_properties(${target} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${KUFCOD_BIN_DIR}
        INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH ON)
    if(ARG_EXPECT_FAIL)
        # Known divergences: one CTest entry that passes while the divergence exists.
        add_test(NAME ${prefix}.known_divergences COMMAND ${target})
        set_tests_properties(${prefix}.known_divergences PROPERTIES WILL_FAIL TRUE)
    else()
        catch_discover_tests(${target} TEST_PREFIX "${prefix}." DISCOVERY_MODE PRE_TEST)
    endif()
endfunction()
