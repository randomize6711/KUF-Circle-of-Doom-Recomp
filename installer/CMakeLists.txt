# The graphical setup (installer/README.md): a GTK 3 program, built on its own (it does not need the recompiled game):
#   cmake -S installer -B build/installer -G Ninja && ninja -C build/installer
# Targets: kuf_setup_core (the tested logic), kuf-setup (the program), kuf_setup_tests (unit tests, no GTK window).
cmake_minimum_required(VERSION 3.25)
project(kuf_setup C CXX)

# Published binaries carry no build-machine paths (__FILE__ relative to the source tree).
add_compile_options("-fmacro-prefix-map=${CMAKE_SOURCE_DIR}/=")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# The GTK window is the Linux setup's; on Windows (M16, llvm-mingw) the core, the local builder and the tests are built
# here and the Windows front end separately.
if(NOT WIN32)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
    pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
    pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
endif()

# The retail XEX key for the setup's and the launchers' decryption (src/core/xex_decrypt.h, docs/RETAIL_FILES.md
# "Decision of 2026-11-08 (2)"). ON (the default, and every release since that decision): read from the pinned SDK's
# git history (content tools/apply_sdk_patches.sh deletes from the working tree) into a header in the build folder;
# the repository never holds it. Linked only into kuf-setup, kuf-launcher and the setup tests (src/retail_key.cpp),
# never into the game package (kuf-localbuild, the runtime). ON without the SDK's history is an error; OFF builds
# without decryption (a retail dump then needs a separate unencrypted default.xex).
option(KUF_SETUP_RETAIL_DECRYPTION "Decrypt a retail default.xex in the setup and the launchers" ON)
set(KUF_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../thirdparty/rexglue-sdk")
set(_kuf_key_have 0)
set(_kuf_key_bytes "0")
if(KUF_SETUP_RETAIL_DECRYPTION)
    find_package(Git QUIET)
    if(GIT_FOUND)
        execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${KUF_SDK_DIR}" show HEAD:src/system/xex_module.cpp
                        OUTPUT_VARIABLE _kuf_src RESULT_VARIABLE _kuf_rc ERROR_QUIET)
        if(_kuf_rc EQUAL 0 AND _kuf_src MATCHES "xe_xex2_retail_key\\[16\\] *= *\\{([^}]*)\\}")
            string(REGEX MATCHALL "0x[0-9A-Fa-f][0-9A-Fa-f]" _kuf_list "${CMAKE_MATCH_1}")
            list(LENGTH _kuf_list _kuf_n)
            if(_kuf_n EQUAL 16)
                string(JOIN ", " _kuf_key_bytes ${_kuf_list})
                set(_kuf_key_have 1)
            endif()
        endif()
        unset(_kuf_src)
    endif()
endif()
file(WRITE "${CMAKE_BINARY_DIR}/generated/kuf_retail_key.h.new"
     "// Generated by installer/CMakeLists.txt: never commit.\n#pragma once\n#include <cstdint>\n"
     "#define KUF_SETUP_HAS_RETAIL_KEY ${_kuf_key_have}\n"
     "inline constexpr uint8_t kRetailXexKeyBytes[] = {${_kuf_key_bytes}};\n")
configure_file("${CMAKE_BINARY_DIR}/generated/kuf_retail_key.h.new" "${CMAKE_BINARY_DIR}/generated/kuf_retail_key.h"
               COPYONLY)
file(REMOVE "${CMAKE_BINARY_DIR}/generated/kuf_retail_key.h.new")
unset(_kuf_key_bytes)
unset(_kuf_list)
if(KUF_SETUP_RETAIL_DECRYPTION AND NOT _kuf_key_have)
    message(FATAL_ERROR "KUF_SETUP_RETAIL_DECRYPTION=ON but the retail key was not found in the history of "
                        "${KUF_SDK_DIR} (a full clone of the pinned SDK is needed); -DKUF_SETUP_RETAIL_DECRYPTION=OFF "
                        "builds without decryption")
endif()
if(_kuf_key_have)
    message(STATUS "kuf-setup: retail default.xex decryption on")
else()
    message(STATUS "kuf-setup: retail default.xex decryption off")
endif()

# The core shared by every front end (the Linux setup, the portable launchers, the Windows setup): the game files,
# disc images, the local build, the portable edition. kuf_setup_install: the Linux setup's installation, Steam
# shortcut and system check (Windows: not yet ported, docs/WINDOWS.md).
add_library(kuf_setup_core STATIC
    src/core/util.cpp
    src/core/game_data.cpp
    src/core/xiso.cpp
    src/core/xex_decrypt.cpp
    src/core/vdf.cpp
    src/core/localbuild.cpp
    src/core/platform.cpp
    src/core/sha256.cpp
    src/core/portable.cpp
    src/core/update/sha512.cpp
    src/core/update/ed25519.cpp
    src/core/update/sshsig.cpp
    src/core/update/json.cpp
    src/core/update/zip.cpp
    src/core/update/updater.cpp
    src/core/update/http_win.cpp src/core/update/http_wininet.cpp)
target_include_directories(kuf_setup_core PUBLIC src)
# Vulkan's headers only (the loader is opened at run time): the SDK's pinned copy.
target_include_directories(kuf_setup_core PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/../thirdparty/rexglue-sdk/thirdparty/vulkan-headers/include")
target_link_libraries(kuf_setup_core PUBLIC ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(kuf_setup_core PUBLIC ole32 uuid shell32 winhttp wininet)  # shell links, known folders; updates
endif()
# No glib in the core (M16: shared with the Windows setup and the portable launchers); the GTK window has it.
target_compile_options(kuf_setup_core PRIVATE -Wall -Wextra)
if(NOT WIN32)
    add_library(kuf_setup_install STATIC src/core/steam.cpp src/core/install.cpp src/core/system_check.cpp)
    target_link_libraries(kuf_setup_install PUBLIC kuf_setup_core ${CMAKE_DL_LIBS})
    target_include_directories(kuf_setup_install PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/../thirdparty/rexglue-sdk/thirdparty/vulkan-headers/include")
    target_compile_options(kuf_setup_install PRIVATE -Wall -Wextra)
endif()
# The XEX decryption routine itself is compiled in only with the key, or for its tests with
# KUF_SETUP_DECRYPTION_CODE_FOR_TESTS (a test key, never the retail one).
option(KUF_SETUP_DECRYPTION_CODE_FOR_TESTS "Compile the XEX decryption routine without the key (tests only)" OFF)
if(KUF_SETUP_RETAIL_DECRYPTION OR KUF_SETUP_DECRYPTION_CODE_FOR_TESTS)
    target_compile_definitions(kuf_setup_core PUBLIC KUF_SETUP_XEX_DECRYPTION=1)
else()
    target_compile_definitions(kuf_setup_core PUBLIC KUF_SETUP_XEX_DECRYPTION=0)
endif()

if(NOT WIN32)
    add_executable(kuf-setup src/gui/main.cpp src/gui/unattended.cpp src/retail_key.cpp)
    target_include_directories(kuf-setup PRIVATE "${CMAKE_BINARY_DIR}/generated")
    target_link_libraries(kuf-setup PRIVATE kuf_setup_install PkgConfig::GTK3 PkgConfig::FONTCONFIG PkgConfig::GLIB)
    target_compile_options(kuf-setup PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
endif()

# The portable launcher (M16, docs/PORTABLE.md): one SDL3 + Dear ImGui program for Linux and Windows, the logic in
# core/portable. SDL3 and ImGui from the pinned SDK, built here as static libraries with SDL's 2D renderer (the game's
# build leaves that out).
option(KUF_SETUP_LAUNCHER "Build the portable launcher (SDL3, Dear ImGui)" ON)
if(KUF_SETUP_LAUNCHER)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
    set(SDL_RENDER ON CACHE BOOL "" FORCE)
    set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
    set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
    set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
    set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
    set(SDL_POWER OFF CACHE BOOL "" FORCE)
    set(SDL_GPU OFF CACHE BOOL "" FORCE)
    set(SDL_RENDER_GPU OFF CACHE BOOL "" FORCE)
    set(SDL_RENDER_VULKAN OFF CACHE BOOL "" FORCE)
    enable_language(C)
    add_subdirectory(${KUF_SDK_DIR}/thirdparty/sdl3 sdl3 EXCLUDE_FROM_ALL)
    set(KUF_IMGUI ${KUF_SDK_DIR}/thirdparty/imgui)
    add_library(kuf_imgui STATIC ${KUF_IMGUI}/imgui.cpp ${KUF_IMGUI}/imgui_draw.cpp ${KUF_IMGUI}/imgui_tables.cpp
        ${KUF_IMGUI}/imgui_widgets.cpp ${KUF_IMGUI}/backends/imgui_impl_sdl3.cpp
        ${KUF_IMGUI}/backends/imgui_impl_sdlrenderer3.cpp)
    target_include_directories(kuf_imgui PUBLIC ${KUF_IMGUI} ${KUF_IMGUI}/backends)
    target_link_libraries(kuf_imgui PUBLIC SDL3::SDL3-static)
    add_executable(kuf-launcher WIN32 src/frontend/launcher.cpp src/retail_key.cpp)
    target_include_directories(kuf-launcher PRIVATE "${CMAKE_BINARY_DIR}/generated")
    target_link_libraries(kuf-launcher PRIVATE kuf_setup_core kuf_imgui)
    target_compile_options(kuf-launcher PRIVATE -Wall -Wextra)
    if(WIN32)
        set_target_properties(kuf-launcher PROPERTIES OUTPUT_NAME "Play KUF")
        target_link_options(kuf-launcher PRIVATE -static-libgcc -static)  # one file: no libc++.dll beside it
    endif()
endif()

# The local build without a window, for the updater (docs/LOCAL_BUILD.md).
add_executable(kuf-localbuild src/localbuild_main.cpp)
target_link_libraries(kuf-localbuild PRIVATE kuf_setup_core)

# The updater of the portable edition and the Windows installation (M17G, core/update/updater.h).
add_executable(kuf-update src/update_main.cpp)
target_link_libraries(kuf-update PRIVATE kuf_setup_core)

# Windows: the UTF-8 process code page (cmake/windows/utf8.manifest), so paths outside the ANSI code page work.
if(WIN32)
    enable_language(RC)
    include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/windows/utf8_manifest.cmake)
    kufcod_utf8_manifest(kuf-localbuild)
    kufcod_utf8_manifest(kuf-update)
    if(TARGET kuf-launcher)
        kufcod_utf8_manifest(kuf-launcher)
    endif()
endif()

# Unit tests (Catch2 from the pinned SDK, as the game's tests). Linux only: they cover the Linux-only parts too
# (kuf_setup_install: Steam, the installation) and their helpers are POSIX (tests/sandbox.h); a Windows build has no
# test targets, and the shared core (kuf_setup_core) is tested on Linux (docs/KNOWN_ISSUES.md K81).
if(NOT WIN32 AND EXISTS "${KUF_SDK_DIR}/thirdparty/catch2/CMakeLists.txt")
    add_subdirectory(${KUF_SDK_DIR}/thirdparty/catch2 catch2 EXCLUDE_FROM_ALL)
    add_executable(kuf_setup_tests tests/test_core.cpp tests/test_steam.cpp tests/test_install.cpp
        tests/test_retail.cpp tests/test_portable.cpp tests/test_update.cpp src/retail_key.cpp)
    target_include_directories(kuf_setup_tests PRIVATE "${CMAKE_BINARY_DIR}/generated")
    target_link_libraries(kuf_setup_tests PRIVATE kuf_setup_install Catch2::Catch2WithMain)
    target_compile_definitions(kuf_setup_tests PRIVATE
        KUF_SETUP_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
    # A full installation (installer/README.md "Testing").
    add_executable(kuf_setup_e2e tests/e2e_install.cpp src/retail_key.cpp)
    target_include_directories(kuf_setup_e2e PRIVATE "${CMAKE_BINARY_DIR}/generated")
    target_link_libraries(kuf_setup_e2e PRIVATE kuf_setup_install)
    enable_testing()
    add_test(NAME setup.unit COMMAND kuf_setup_tests)
endif()
